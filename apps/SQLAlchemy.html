1. Session Lifecycle

Create Session Factory

engine = create_async_engine(
    DATABASE_URL,
    echo=True,
)

Session = async_sessionmaker(
    engine,
    expire_on_commit=False,
)

Create Session

async with Session() as session:
    ...

Transaction Session

async with Session.begin() as session:
    ...

Automatically:

BEGIN
COMMIT
ROLLBACK (on exception)

2. Create Operations

add()

Add single object.

user = User(
    name="Vansh",
    email="vansh@example.com",
)

session.add(user)

await session.commit()

add_all()

Add multiple objects.

session.add_all([
    user1,
    user2,
    user3,
])

flush()

Push SQL to DB without commit.

session.add(user)

await session.flush()

print(user.id)

Useful when primary key is needed immediately.

refresh()

Reload latest database state.

await session.refresh(user)

3. Read Operations

get()

Fastest way to fetch by Primary Key.

user = await session.get(
    User,
    user_id,
)

select()

Basic query.

stmt = select(User)

result = await session.execute(stmt)

users = result.scalars().all()

where()

Filtering.

stmt = select(User).where(
    User.email == email
)

Multiple Conditions

stmt = select(User).where(
    User.name == "John",
    User.is_active.is_(True),
)

order_by()

stmt = (
    select(User)
    .order_by(User.created_at.desc())
)

limit()

stmt = select(User).limit(10)

offset()

stmt = (
    select(User)
    .offset(20)
    .limit(10)
)

4. Update Operations

ORM Update (Recommended)

user = await session.get(
    User,
    user_id,
)

user.name = "Updated Name"

await session.commit()

Multiple Fields

user.name = payload.name
user.email = payload.email
user.is_active = payload.is_active

Dynamic Update

updates = payload.model_dump(
    exclude_unset=True
)

for field, value in updates.items():
    setattr(
        user,
        field,
        value,
    )

SQL Update

stmt = (
    update(User)
    .where(User.id == user_id)
    .values(name="Updated")
)

await session.execute(stmt)

await session.commit()

Used mostly for bulk updates.

5. Delete Operations

ORM Delete

user = await session.get(
    User,
    user_id,
)

await session.delete(user)

await session.commit()

Triggers ORM cascade rules.

SQL Delete

stmt = (
    delete(User)
    .where(User.id == user_id)
)

await session.execute(stmt)

await session.commit()

Does NOT trigger ORM cascades.

PostgreSQL RETURNING

stmt = (
    delete(User)
    .where(User.id == user_id)
    .returning(User.id)
)

result = await session.execute(stmt)

deleted_ids = result.scalars().all()

commit()

await session.commit()

Persist changes.

rollback()

await session.rollback()

Undo transaction.

begin()

async with session.begin():
    ...

Automatic transaction management.

7. Relationship Loading

Lazy Loading

user.addresses

Avoid in Async SQLAlchemy.

selectinload()

Most commonly used.

stmt = (
    select(User)
    .options(
        selectinload(
            User.addresses
        )
    )
)

joinedload()

stmt = (
    select(User)
    .options(
        joinedload(
            User.addresses
        )
    )
)

Single JOIN query.

subqueryload()

stmt = (
    select(User)
    .options(
        subqueryload(
            User.addresses
        )
    )
)

Less common.

8. Result Processing

scalars().all()

result = await session.execute(stmt)

users = result.scalars().all()

Most common.

scalar_one()

Exactly one record.

user = result.scalar_one()

Raises exception if:

0 rows
More than 1 row

scalar_one_or_none()

user = result.scalar_one_or_none()

Very common in APIs.

scalars().first()

user = result.scalars().first()

Returns first result.

all()

rows = result.all()

Returns tuples.

9. Joins

Inner Join

stmt = (
    select(Product)
    .join(Product.category)
)

Outer Join

stmt = (
    select(Product)
    .outerjoin(Product.category)
)

Multi Join

stmt = (
    select(Order)
    .join(Order.customer)
    .join(Order.items)
)

10. Aggregation

Count

stmt = select(
    func.count(User.id)
)

Sum

stmt = select(
    func.sum(Order.total)
)

Average

stmt = select(
    func.avg(Product.price)
)

Max

stmt = select(
    func.max(Product.price)
)

Min

stmt = select(
    func.min(Product.price)
)

Group By

stmt = (
    select(
        Product.category_id,
        func.count(Product.id)
    )
    .group_by(Product.category_id)
)

11. Advanced Session Methods

merge()

Attach detached object.

await session.merge(user)

Rare.

expunge()

Remove object from session.

session.expunge(user)

expire()

Mark object for reload.

session.expire(user)

expire_all()

session.expire_all()

12. Common Relationship Configurations

One-To-Many

class User(Base):

    addresses = relationship(
        "Address",
        back_populates="user",
        cascade="all, delete-orphan",
    )

Many-To-One

class Address(Base):

    user = relationship(
        "User",
        back_populates="addresses",
    )

Many-To-Many

tags = relationship(
    "Tag",
    secondary=product_tags,
    back_populates="products",
)

13. Production Patterns
